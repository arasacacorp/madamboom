    15→/* ─── Seeded pseudo-random for deterministic SSR ─── */
    16→function seededRandom(seed: number) {
    17→  const x = Math.sin(seed + 1) * 10000
    18→  return x - Math.floor(x)
    19→}

    21→/* ─── Floating Particles ─── */
    22→function SectionParticles() {
    23→  const particles = Array.from({ length: 8 }, (_, i) => ({
    24→    id: i,
    25→    size: seededRandom(i * 11 + 500) * 3 + 1,
    26→    left: seededRandom(i * 13 + 510) * 100,
    27→    top: seededRandom(i * 17 + 520) * 100,
    28→    opacity: seededRandom(i * 19 + 530) * 0.12 + 0.03,
    29→    duration: seededRandom(i * 23 + 540) * 16 + 14,
    30→    delay: seededRandom(i * 29 + 550) * 10,
    31→    driftX: seededRandom(i * 31 + 560) * 50 - 25,
    32→    driftY: -(seededRandom(i * 37 + 570) * 80 + 20),
    33→    driftX2: seededRandom(i * 41 + 580) * 40 - 20,
    34→    driftY2: -(seededRandom(i * 43 + 590) * 100 + 30),
    35→  }))
    36→
    37→  return (
    38→    <div className="absolute inset-0 overflow-hidden pointer-events-none" style={{ zIndex: 2 }}>
    39→      {particles.map((p) => (
    40→        <div
    41→          key={p.id}
    42→          className="particle"
    43→          style={{
    44→            width: p.size,
    45→            height: p.size,
    46→            left: `${p.left}%`,
    47→            top: `${p.top}%`,
    48→            '--particle-opacity': p.opacity,
    49→            '--duration': `${p.duration}s`,
    50→            '--delay': `${p.delay}s`,
    51→            '--drift-x': `${p.driftX}px`,
    52→            '--drift-y': `${p.driftY}px`,
    53→            '--drift-x2': `${p.driftX2}px`,
    54→            '--drift-y2': `${p.driftY2}px`,
    55→          } as React.CSSProperties}
    56→        />
    57→      ))}
    58→    </div>
    59→  )
    60→}
