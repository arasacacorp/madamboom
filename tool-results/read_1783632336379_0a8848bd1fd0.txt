    15→    15→/* ─── Seeded pseudo-random for deterministic SSR ─── */
    16→    16→function seededRandom(seed: number) {
    17→    17→  const x = Math.sin(seed + 1) * 10000
    18→    18→  return x - Math.floor(x)
    19→    19→}

    22→    22→function SectionParticles() {
    23→    23→  const particles = Array.from({ length: 8 }, (_, i) => ({
    24→    24→    id: i,
    25→    25→    size: seededRandom(i * 11 + 500) * 3 + 1,
    26→    26→    left: seededRandom(i * 13 + 510) * 100,
    27→    27→    top: seededRandom(i * 17 + 520) * 100,
    28→    28→    opacity: seededRandom(i * 19 + 530) * 0.12 + 0.03,
    29→    29→    duration: seededRandom(i * 23 + 540) * 16 + 14,
    30→    30→    delay: seededRandom(i * 29 + 550) * 10,
    31→    31→    driftX: seededRandom(i * 31 + 560) * 50 - 25,
    32→    32→    driftY: -(seededRandom(i * 37 + 570) * 80 + 20),
    33→    33→    driftX2: seededRandom(i * 41 + 580) * 40 - 20,
